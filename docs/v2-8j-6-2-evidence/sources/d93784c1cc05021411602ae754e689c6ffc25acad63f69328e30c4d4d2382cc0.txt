async function hold(page,pattern,method='GET'){
  await page.evaluate(`(()=>{const previous=window.fetch;let used=false;window.__heldExport=false;window.fetch=async(url,options={})=>{const response=await previous(url,options);if(!used&&String(url).includes(${JSON.stringify(pattern)})&&(options.method??'GET')===${JSON.stringify(method)}){used=true;window.__heldExport=true;await new Promise(r=>window.__releaseExport=r);}return response;};})()`);
}
