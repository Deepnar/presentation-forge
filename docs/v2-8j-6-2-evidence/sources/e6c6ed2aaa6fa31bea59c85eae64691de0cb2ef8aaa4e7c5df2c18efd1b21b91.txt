test('native export HTTP: shared metadata/render admission survives abort and faults until actual settlement',async t=>{
  const old=process.env.FORGE_HOSTED;process.env.FORGE_HOSTED='0';t.after(()=>{if(old===undefined)delete process.env.FORGE_HOSTED;else process.env.FORGE_HOSTED=old;});
  let release,held=true,entered=0,mode='normal';const gate=new Promise(r=>release=r);
  const result=b=>({ok:true,snapshot:{projectId:b.id,projectRevision:0,sourceGeneration:0,sourceRevision:0},citationCount:0});
  const app=express();installNativeExportHeaders(app);installNativeExportParser(app);registerNativeExportRoutes(app,{service:async()=>({
    inspect:async b=>{entered++;if(held)await gate;if(mode==='fault')throw Error('PRIVATE-TOKEN');return result(b);},
    render:async b=>{entered++;if(held)await gate;return {...result(b),bytes:new Uint8Array(mode==='oversize'?NATIVE_EXPORT_BYTES+1:10)};},
  })});
  const server=app.listen(0,'127.0.0.1');await new Promise(r=>server.once('listening',r));t.after(async()=>{release();server.closeAllConnections();await new Promise(r=>server.close(r));});const url='http://127.0.0.1:'+server.address().port+'/api/v2/projects/p/native-export';
  const a=new AbortController(),first=fetch(url,{signal:a.signal,headers:{Authorization:'Bearer '+f.token}}).catch(e=>e);
  while(entered<1)await new Promise(r=>setImmediate(r));assert.equal((await fetch(url,{headers:{Authorization:'Bearer '+f.token}})).status,429);
  const second=fetch(url,{method:'POST',headers:{Authorization:'Bearer '+f.otherToken,'Content-Type':'application/json'},body:JSON.stringify({...q,projectRevision:0,sourceRevision:0})});
  while(entered<2)await new Promise(r=>setImmediate(r));a.abort();await first;assert.equal((await fetch(url)).status,429);
  held=false;release();assert.equal((await second).status,200);assert.equal((await fetch(url)).status,200);
  mode='fault';const fault=await fetch(url);assert.equal(fault.status,503);assert.equal((await fault.text()).includes('PRIVATE'),false);
  mode='oversize';const oversized=await fetch(url,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(q)});assert.equal(oversized.status,413);assert.equal(oversized.headers.has('content-disposition'),false);
  mode='normal';assert.equal((await fetch(url)).status,200);
});
